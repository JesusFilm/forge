import {chromium} from '/home/tataihono/.codex/worktrees/5dd0/forge/apps/shorts-worker/node_modules/playwright/index.mjs';
import {readFile,writeFile} from 'node:fs/promises';
const {url}=JSON.parse(await readFile('/tmp/forge-studio-456-prep/final-browser-project.json','utf8'));
const browser=await chromium.launch({executablePath:'/usr/bin/google-chrome',headless:true,args:['--host-resolver-rules=MAP images.jesusfilm.org ~NOTFOUND']});
const context=await browser.newContext({viewport:{width:1440,height:1000}});
await context.request.get('http://studio456.localhost:3458/local-fixture-login',{maxRedirects:0});
const page=await context.newPage(),events=[],errors=[];page.on('pageerror',e=>errors.push(e.stack));
page.on('response',async r=>{if(r.url().endsWith('/api/studio/preview')){const b=await r.json().catch(()=>({}));events.push({status:r.status(),error:b.error,transferredBytes:b.transferredBytes,at:Date.now()});}});
try{
 await page.goto(url);
 await page.getByRole('button',{name:/^JESUS/}).click();
 await page.getByLabel('Source in (seconds)',{exact:true}).fill('28.6');
 await page.getByLabel('Source out (seconds)',{exact:true}).fill('30.6');
 await page.getByRole('button',{name:'Add footage',exact:true}).click();
 await page.getByRole('dialog').waitFor({state:'hidden',timeout:45000});
 await page.waitForFunction(()=>document.querySelector('.nle-preview')?.getAttribute('data-preview-ready')==='true',{},{timeout:120000});
 const preview=page.frameLocator('iframe[title="Live composition preview"]');
 await preview.locator('video').waitFor({timeout:120000});
 const iframe=page.frames().find(f=>f.url().startsWith('http://127.0.0.1:3460/'));
 await iframe.waitForFunction(()=>{const v=document.querySelector('video');return v&&v.readyState>=2},{},{timeout:120000});
 const media=await iframe.evaluate(()=>{const v=document.querySelector('video');return {currentTime:v.currentTime,width:v.videoWidth,height:v.videoHeight,readyState:v.readyState}});
 await page.getByRole('button',{name:'Save',exact:true}).click();await page.getByRole('status').filter({hasText:/Saved/}).waitFor();
 await page.screenshot({path:'/tmp/forge-studio-456-prep/editor-source-final.png'});
 await writeFile('/tmp/forge-studio-456-prep/source-browser-final.json',JSON.stringify({url,media,events,errors},null,2));console.log({media,events,errors});
}catch(e){console.error(e);console.log(await page.locator('body').innerText(),events,errors);await page.screenshot({path:'/tmp/forge-studio-456-prep/source-failure-final.png'});process.exitCode=1}finally{await browser.close()}
