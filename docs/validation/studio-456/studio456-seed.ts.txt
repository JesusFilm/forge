import {readFileSync} from 'node:fs'
import {PrismaClient} from '@prisma/client'
async function main(){
 const url=process.env.DATABASE_URL!
 if(url!=='postgresql://studio456@127.0.0.1:55456/forge_studio_456_browser')throw new TypeError('Own disposable DB required')
 const db=new PrismaClient()
 try{
 const video=JSON.parse(readFileSync('/tmp/forge-studio-456-reproduction/public-catalog.json','utf8')).data.videoBySlug,dub=video.preferredPlayableDub,lang=dub.language
 await db.user.upsert({where:{id:'mock-manager-1'},create:{id:'mock-manager-1',email:'manager@forge.test',name:'Studio local operator',role:'ADMIN',managerMembership:{create:{role:'OPERATOR'}}},update:{}})
 await db.language.upsert({where:{id:lang.id},create:lang,update:{}})
 await db.video.upsert({where:{id:video.id},create:{id:video.id,coreId:video.id,slug:video.slug,noIndex:false,restrictViewPlatforms:video.restrictViewPlatforms,locales:{create:{status:'PUBLISHED',title:'JESUS',locale:'en'}}},update:{}})
 await db.videoEdition.upsert({where:{id:dub.videoEdition.id},create:{id:dub.videoEdition.id,coreId:dub.videoEdition.id,name:'JESUS edition'},update:{}})
 await db.videoDub.upsert({where:{id:dub.id},create:{id:dub.id,coreId:dub.id,videoId:video.id,videoEditionId:dub.videoEdition.id,languageId:lang.id,published:dub.published,downloadable:dub.downloadable,lengthInMilliseconds:BigInt(dub.lengthInMilliseconds),hls:dub.hls,downloads:{create:dub.downloads}},update:{}})
 const track=dub.videoEdition.subtitles.find((t:any)=>t.id==='1a1c8171-c89e-4f91-9e26-86fc1fe7125e')

 console.log('Seeded local operator and exact public JESUS catalog identity')
 }finally{await db.$disconnect()}
}
main().catch(e=>{console.error(e);process.exitCode=1})
