import os,subprocess,socket,json,time,hashlib,re
from pathlib import Path
base=Path('/home/tataihono/.local/share/forge/studio-462-release')
pg=Path('/usr/lib/postgresql/18/bin')
url='postgresql://tataihono@127.0.0.1:55462/forge_studio_462_test'
env={k:os.environ[k] for k in ['PATH','HOME'] if k in os.environ}
env.update(TMPDIR=str(base/'tmp'),DATABASE_URL=url,CI='1',NEXT_TELEMETRY_DISABLED='1',DD_TRACE_ENABLED='false',OTEL_SDK_DISABLED='true',REDIS_URL='redis://127.0.0.1:54629',ADMIN_SESSION_SECRET='studio462-local-fixture-secret-at-least-32',AUTH_ISSUER_URL='http://127.0.0.1:55469',AUTH_ADMIN_CLIENT_ID='studio462-fixture')
results=[]
def run(name,args):
 start=time.monotonic()
 with (base/'logs'/f'{name}.log').open('w') as f:r=subprocess.run(args,env=env,stdout=f,stderr=subprocess.STDOUT)
 results.append(dict(name=name,exit=r.returncode,seconds=round(time.monotonic()-start,3)))
 (base/'evidence/correction-results.json').write_text(json.dumps(results,indent=2)+'\n')
 print(name,r.returncode,flush=True)
 if r.returncode:raise RuntimeError(name)
try:
 subprocess.run(['ip','link','set','lo','up'],check=True)
 for port in [5432,6379]:
  with socket.socket() as s:
   s.settimeout(.2);assert s.connect_ex(('127.0.0.1',port))!=0
 with socket.socket() as s:s.bind(('127.0.0.1',55462))
 assert not (base/'pgdata/postmaster.pid').exists()
 run('pg-restart',[str(pg/'pg_ctl'),'-D',str(base/'pgdata'),'-l',str(base/'logs/postgres.log'),'-o',f'-h 127.0.0.1 -p 55462 -k {base}/tmp','-w','start'])
 run('studio-drift-green',['pnpm','--filter','@forge/admin','exec','prisma','migrate','diff','--from-url',url,'--to-schema-datamodel','prisma/schema.prisma','--script'])
 diff=(base/'logs/studio-drift-green.log').read_text()
 studio=[stmt for stmt in diff.split(';') if re.search(r'"(?:studio_|content_pack)',stmt)]
 assert not studio,studio
 (base/'evidence/studio-drift-green.json').write_text(json.dumps(dict(studioStatements=studio,qualification='Unrelated inherited schema differences remain. Diff SQL was never applied; reviewed migrations unchanged.'),indent=2)+'\n')
 run('prisma-generate-green',['pnpm','--filter','@forge/admin','db:generate'])
 run('schema-print-green',['pnpm','--filter','@forge/admin','schema:print'])
 run('graphql-generate-green',['pnpm','--filter','@forge/admin-graphql','generate'])
 before=json.loads((base/'evidence/generated-before.json').read_text())
 after={p:hashlib.sha256(Path(p).read_bytes()).hexdigest() for p in before}
 (base/'evidence/generated-after.json').write_text(json.dumps(dict(before=before,after=after,identical=before==after),indent=2)+'\n')
 run('schema-consumer-tests',['pnpm','--filter','@forge/admin','exec','vitest','run','src/graphql/schema.test.ts','src/services/studio-authoring/scheduled-publication-adapter.test.ts','--maxWorkers=1','--no-file-parallelism'])
 run('admin-typecheck-green',['pnpm','--filter','@forge/admin','typecheck'])
 run('graphql-typecheck-green',['pnpm','--filter','@forge/admin-graphql','typecheck'])
finally:
 if (base/'pgdata/postmaster.pid').exists():run('pg-stop-green',[str(pg/'pg_ctl'),'-D',str(base/'pgdata'),'-m','fast','-w','stop'])
