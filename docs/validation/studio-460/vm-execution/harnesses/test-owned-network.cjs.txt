const net = require('node:net');
const owned = new Set([55460]);
const listen = net.Server.prototype.listen;
net.Server.prototype.listen = function (...args) {
  if (typeof args[0] === 'object') {
    if (args[0].path || (args[0].host && !['localhost','127.0.0.1','::1'].includes(args[0].host))) throw new Error('STUDIO460_TEST_LISTEN_DENIED');
    args[0] = {...args[0], host: args[0].host || '127.0.0.1'};
  } else {
    if (typeof args[1] === 'string' && !['localhost','127.0.0.1','::1'].includes(args[1])) throw new Error('STUDIO460_TEST_LISTEN_DENIED');
    if (typeof args[1] !== 'string') args.splice(1,0,'127.0.0.1');
  }
  this.once('listening', () => { const a = this.address(); if (a && typeof a === 'object') owned.add(a.port); });
  return listen.apply(this, args);
};

function ownedBuildIpc(port) {
  const fs=require('node:fs');
  const base='/home/tataihono/.cache/forge-studio-460-image-worktree';
  if(!['manager','admin'].some(app=>['postcss.js','webpack-loaders.js'].some(entry=>process.argv[1]===base+'/apps/'+app+'/.next/build/'+entry)) || Number(process.argv[2])!==port)return false;
  try {
    const parent='/proc/'+process.ppid;
    const command=fs.readFileSync(parent+'/cmdline','utf8').split('\0');
    if(!command.some(v=>v.startsWith(base+'/')&&v.endsWith('/next/dist/bin/next')) || !command.includes('build'))return false;
    const sockets=new Set(fs.readdirSync(parent+'/fd').flatMap(fd=>{try{const m=fs.readlinkSync(parent+'/fd/'+fd).match(/^socket:\[(\d+)\]$/);return m?[m[1]]:[]}catch{return []}}));
    for(const table of ['tcp','tcp6'])for(const line of fs.readFileSync(parent+'/net/'+table,'utf8').trim().split('\n').slice(1)){
      const fields=line.trim().split(/\s+/), parts=fields[1].split(':');
      if(fields[3]==='0A'&&parseInt(parts[1],16)===port&&['0100007F','00000000000000000000000001000000'].includes(parts[0])&&sockets.has(fields[9])){fs.appendFileSync('/home/tataihono/.cache/forge-studio-460-vm/evidence/build-ipc-accepted.jsonl',JSON.stringify({pid:process.pid,parent:process.ppid,port,local:fields[1],inode:fields[9],entrypoint:process.argv[1]})+'\n');return true;}
    }
  }catch{}
  return false;
}

const connect = net.Socket.prototype.connect;
net.Socket.prototype.connect = function (...args) {
  let first = args[0]; if (Array.isArray(first)) first = first[0];
  const options = typeof first === 'object' ? first : { port: first, host: typeof args[1] === 'string' ? args[1] : 'localhost' };
  const host = options.host || options.hostname || 'localhost';
  if (!['127.0.0.1', 'localhost', '::1'].includes(host) || (!owned.has(Number(options.port)) && !ownedBuildIpc(Number(options.port)))) {if(process.argv[1]?.startsWith('/home/tataihono/.cache/forge-studio-460-image-worktree/apps/')) require('node:fs').appendFileSync('/home/tataihono/.cache/forge-studio-460-vm/evidence/build-ipc-denied.jsonl',JSON.stringify({pid:process.pid,parent:process.ppid,entrypoint:process.argv[1],host,port:Number(options.port)})+'\n');throw new Error('STUDIO460_TEST_ENDPOINT_DENIED');}
  return connect.apply(this, args);
};
