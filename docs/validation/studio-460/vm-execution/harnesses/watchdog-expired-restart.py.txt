import json,os,subprocess,time,uuid
from pathlib import Path
id=uuid.uuid4().hex;state=Path('/var/lib/forge-studio/jobs')/id;state.mkdir(mode=0o700)
slice='forge-studio-j'+id+'.slice';cg=Path('/sys/fs/cgroup/forge.slice/forge-studio.slice')/slice
Path('/run/systemd/system/'+slice).write_text('[Unit]\nStopWhenUnneeded=no\n[Slice]\nCPUQuota=200%\nMemoryMax=2147483648\nMemorySwapMax=0\nTasksMax=128\n')
def run(args):return subprocess.run(args,check=False,capture_output=True,text=True,timeout=5)
run(['/usr/bin/systemctl','daemon-reload']);run(['/usr/bin/systemctl','start',slice]);boot=Path('/proc/sys/kernel/random/boot_id').read_text().strip();deadline=int(time.monotonic()*1000)-1
unit='forge-studio-watchdog-'+id
result=run(['/usr/bin/systemd-run','--unit='+unit,'--property=Type=notify','--property=NotifyAccess=main','--property=Requires='+slice,'--property=After='+slice,'--property=Restart=on-failure','--property=RestartSec=100ms','--property=MemoryMax=32M','--property=TasksMax=8','--property=CPUQuota=5%','/opt/forge-studio/current/vm-watchdog',id,boot,str(deadline),str(cg.stat().st_ino)])
time.sleep(2)
record={'id':id,'unit':unit,'slice':slice,'boot':boot,'deadline':deadline,'inode':str(cg.stat().st_ino),'initialReturnCode':result.returncode,'initialStderr':result.stderr,'state':run(['/usr/bin/systemctl','show',unit,'--property=ActiveState,SubState,NRestarts,Result']).stdout,'freeze':(cg/'cgroup.freeze').read_text()}
(state/'restart-proof.json').write_text(json.dumps(record));print(json.dumps(record))
