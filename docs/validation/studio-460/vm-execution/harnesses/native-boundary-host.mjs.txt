import {mkdir,writeFile,readFile,copyFile,chown,chmod} from 'node:fs/promises'
import {randomUUID} from 'node:crypto'
import {VmJournal} from '/opt/forge-studio/current/vm/journal.mjs'
import {jobIdentity,containerArguments} from '/opt/forge-studio/current/vm/profile.mjs'
import {createJobScope,cancelJobScope,retireJobScope} from '/opt/forge-studio/current/vm/job-scope.mjs'
import {assertLaunchable} from '/opt/forge-studio/current/vm/launch-gate.mjs'
import {runHostCommand} from '/opt/forge-studio/current/vm/host-command.mjs'
import {DockerJobRuntime} from '/opt/forge-studio/current/vm/docker-runtime.mjs'
const now=()=>Number(process.hrtime.bigint()/1000000n),directory='/var/lib/forge-studio/native-boundary-probe-1';await mkdir(directory,{mode:0o700});
const journal=new VmJournal(directory),claim=await journal.claimRequest({poolId:'owned-kernel-probe',workerId:'owned-kernel-probe'});
const id=claim.dispatchId.replaceAll('-',''),identity=jobIdentity(id),image='sha256:1f5545cad34fdfb4945bae4fce19322037366ee43fadf7555dcf3e9cdc61bcbb';
const record=await journal.assignment({...claim,attemptId:'owned-kernel-probe',leaseId:randomUUID(),expiresAt:Date.now()+1200000},{bootId:(await readFile('/proc/sys/kernel/random/boot_id','utf8')).trim(),deadlineMs:now()+20000,renderImage:image,verifyImage:image});
const gate=await createJobScope(id,record,journal,'/opt/forge-studio/releases/d0c00b9a005f5e3bddf1873466f2127d145b350cd450937ec4a71d7d531d8cd8/vm-watchdog');
await mkdir(identity.directory+'/render-input',{mode:0o700});
for(const name of ['probe','child.mjs']){const target=identity.directory+'/render-input/'+name;await copyFile('/home/forge/studio-build/native-boundary/'+name,target);await chown(target,1000,1000);await chmod(target,0o500)}
await chown(identity.directory+'/render-input',1000,1000);await chmod(identity.directory+'/render-input',0o500);
const docker=(args,options={})=>runHostCommand('/usr/bin/docker',args,{timeoutMs:10000,...options});
await journal.beginOperation('render','create');await assertLaunchable(id,gate);
const args=containerArguments({identity,image,deadlineMs:gate.deadlineMs,phase:'render'});
// Diagnostic-only readonly replacement of the child module exercises the same
// baked guard/init/kernel boundary. No product launcher accepts this extra mount.
args.splice(args.indexOf(image),0,'--mount=type=bind,source='+identity.directory+'/render-input/child.mjs,target=/runtime/child.mjs,readonly');
const cid=(await docker(args)).stdout.trim();await journal.created('render',cid);
await writeFile(directory+'/identity.json',JSON.stringify({id,cid,gate,scope:'kernel diagnostic; not a composition or canonical assignment',args})+'\n',{mode:0o600});
const inspection=JSON.parse((await docker(['inspect',cid])).stdout)[0];await writeFile(directory+'/container.json',JSON.stringify(inspection)+'\n',{mode:0o600});
await journal.beginOperation('render','start');await assertLaunchable(id,gate);const started=now();
try{const result=await docker(['start','--attach',cid],{timeoutMs:Math.max(1,gate.deadlineMs-now()+1000)});await writeFile(directory+'/output.log',result.stdout+'\n'+result.stderr);console.log(JSON.stringify({exit:'success',elapsedMs:now()-started}))}
catch(error){console.log(JSON.stringify({exit:'bounded-non-success',elapsedMs:now()-started,clientStopped:error.clientStopped}));}
finally{
 await writeFile(directory+'/final-container.json',(await docker(['inspect',cid])).stdout);
 await cancelJobScope(id);const end=now()+2000;await new DockerJobRuntime({identity,journal,gate}).reconcile(record.binding,end);await retireJobScope(id,journal,end);
 console.log(JSON.stringify({id,retired:true}));
}
