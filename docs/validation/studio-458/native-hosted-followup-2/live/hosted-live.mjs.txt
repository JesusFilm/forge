import {existsSync} from 'node:fs';
import {chromium} from '/home/tataihono/.codex/worktrees/8b33/forge/apps/shorts-worker/node_modules/playwright/index.mjs';
import {readFile,writeFile} from 'node:fs/promises';

const browser=await chromium.launch({executablePath:'/usr/bin/google-chrome',headless:true});

const button=n=>page.getByRole('button',{name:n,exact:true});
try{
 await context.request.get('http://studio458.localhost:3589/local-fixture-login',{maxRedirects:0});
 for(const c of proposal.cases){if(existsSync('/tmp/forge-studio-458-prep/creative-followup-2/live/STOP.json'))throw Error('Batch stopped; no replacement or next run');
  await page.goto('http://studio458.localhost:3588/dashboard/shorts/'+c.project.projectId,{waitUntil:'domcontentloaded'});
  await button('Generate scripts').click();

  await page.getByLabel('Explicitly generate these',{exact:false}).check();

  await button('Generate script, composition and source preview').click();

  const admitted=events.trim().split('\n').map(JSON.parse).find(e=>e.event?.type==='admitted')?.event?.attemptId;
  if(admitted&&!events.includes('target-error')){
   await button('Load retained generation attempts').click();
   const attempts=page.getByRole('combobox',{name:/^Attempt/});
   await attempts.locator('option[value="'+admitted+'"]').waitFor({state:'attached'});
   await attempts.selectOption(admitted);
   const preview=button('Preview proposed composition');
   try{await preview.waitFor({timeout:5000});await preview.click();await page.getByRole('dialog',{name:'Generate scripts and compositions',exact:true}).locator('.nle-preview[data-preview-ready="true"]').waitFor({timeout:30000});out.at(-1).previewReady=true;}catch(e){out.at(-1).previewIssue=e.message;}

  out.at(-1).visibleText=await page.getByRole('dialog',{name:'Generate scripts and compositions',exact:true}).innerText();
  await page.screenshot({path:'/tmp/forge-studio-458-prep/creative-followup-2/live/live-'+c.case+'.png',fullPage:true});

 }
 if(errors.length)throw new Error(errors.join(';'));
 console.log(JSON.stringify({nativeRuns:out.length,pageErrors:errors,paidRequests:"See durable LIVE ledger; no retries"}));
}finally{await writeFile('/tmp/forge-studio-458-prep/creative-followup-2/live/hosted-live-evidence.json',JSON.stringify({out,pageErrors:errors},null,2));await browser.close();}
