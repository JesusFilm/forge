import {test} from 'node:test';import assert from 'node:assert/strict';import {mkdtempSync,rmSync} from 'node:fs';import {tmpdir} from 'node:os';import {join} from 'node:path';import {withBatchStop} from './batch-stop.mjs';
test('a rejected dispatch durably blocks another run after restart',async()=>{const dir=mkdtempSync(join(tmpdir(),'studio458-stop-')),path=join(dir,'STOP.json');let calls=0;try{const first=withBatchStop(path,async()=>{calls++;throw Error('Preflight or ambiguous failure')});await assert.rejects(first());const restarted=withBatchStop(path,async()=>{calls++});await assert.rejects(restarted(),/Batch stopped/);assert.equal(calls,1)}finally{rmSync(dir,{recursive:true})}});
