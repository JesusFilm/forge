import {readFile,writeFile} from 'node:fs/promises'
import {createHash} from 'node:crypto'
import {resolveInstructionBlocks} from '@mastra/editor'
import {studioCreativeDefaults} from '../src/services/studio-authoring/defaults'
const out='/tmp/forge-studio-458-prep/creative-followup-2'
const previous=JSON.parse(await readFile('../../docs/validation/studio-458/native-hosted-proposal/corrected-readonly-proposal.body','utf8'))
const old=previous.frozenNativeInstructions
const split=old.effective.lastIndexOf('\n\nKeep source evidence separate')
if(split<0)throw new Error('Expected exact retained source block')
const block=old.effective.slice(split+2)
const hash=(s:string)=>createHash('sha256').update(s).digest('hex')
if(hash(block)!==old.blockDigest)throw new Error('Retained block digest mismatch')
const instructions=[{type:'text' as const,content:studioCreativeDefaults},{type:'prompt_block_ref' as const,id:'studio-authoring-voice'}]
const resolved=await resolveInstructionBlocks([{type:'text',content:studioCreativeDefaults},{type:'prompt_block',content:block}],{language:'en'})
const proposed={status:'CALCULATED_NOT_SAVED_OR_ACTIVATED',agentVersionId:null,agentDigest:hash(JSON.stringify(instructions)),blockVersionId:old.blockVersionId,blockDigest:hash(block),digest:hash(resolved),effective:resolved,instructions,context:{language:'en'},utf8Bytes:Buffer.byteLength(resolved)}
await writeFile(out+'/proposed-instructions.json',JSON.stringify(proposed,null,2)+'\n')
await writeFile(out+'/proposed-agent.body',studioCreativeDefaults)
await writeFile(out+'/proposed-effective.body',resolved)
await writeFile(out+'/retained-source-block.body',block)
await writeFile(out+'/previous-agent.body',old.effective.slice(0,split))
console.log(JSON.stringify({agentDigest:proposed.agentDigest,effectiveDigest:proposed.digest,utf8Bytes:proposed.utf8Bytes,blockUnchanged:true,nativeMutations:0,providerCalls:0}))
