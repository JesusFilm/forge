import {test} from 'node:test';import assert from 'node:assert/strict';
import {restrictToolFetch,allowedToolActions} from './tool-boundary.mjs';
test('execution boundary never forwards capture, registration, edits or media/provider actions',async()=>{
 let forwarded=0;const fetch=restrictToolFetch(async()=>{forwarded++;return Response.json({result:true})});
 for(const action of ['capture','asset-upload','registerComponent','apply','narration-complete','experiment-request','publish','render','unknown'])assert.equal((await fetch('http://127.0.0.1:3587/api/studio/tools',{method:'POST',body:JSON.stringify({action})})).status,403);
 assert.equal(forwarded,0);
 for(const action of allowedToolActions)assert.equal((await fetch('http://127.0.0.1:3587/api/studio/tools',{method:'POST',body:JSON.stringify({action})})).status,200);
 assert.equal(forwarded,allowedToolActions.size);
});
