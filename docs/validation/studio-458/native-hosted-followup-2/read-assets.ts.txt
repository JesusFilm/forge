import {PrismaClient} from '@prisma/client'
import {readFile,writeFile} from 'node:fs/promises'
import {deepStrictEqual} from 'node:assert'
import {createHash} from 'node:crypto'
import {StudioAuthoringService} from '../src/services/studio-authoring'
import {StudioAssetService} from '../src/services/studio-authoring/assets'
import {ContentPackService} from '../src/services/studio-authoring/packs'
import {studioAssetReferenceSchema} from '@forge/studio-contracts'
async function main(){
 if(process.env.DATABASE_URL!=='postgresql://tataihono@127.0.0.1:55458/forge_studio_458_test')throw new Error('Own DB only')
 const db=new PrismaClient(),u={id:'mock-manager-1',role:'ADMIN' as const},p=JSON.parse(await readFile('../../docs/validation/studio-458/native-hosted-proposal/corrected-readonly-proposal.body','utf8')),refs=new Map<string,any>(),checks=[]
 const collect=(v:any)=>{if(!v||typeof v!=='object')return;const ref=studioAssetReferenceSchema.safeParse(v);if(ref.success){refs.set(ref.data.versionId,ref.data);return}for(const child of Object.values(v))collect(child)}
 for(const c of p.cases){const project=await new StudioAuthoringService(db).read(u,c.project.projectId);deepStrictEqual(project.document,c.project.document);deepStrictEqual(project.revision,c.project.revision);const pack=await new ContentPackService(db).read(u,c.pack.id);deepStrictEqual(JSON.parse(JSON.stringify(pack)),c.pack);collect(project);collect(pack);checks.push({projectId:project.projectId,revision:project.revision,packId:pack.id})}
 const assets=[];for(const ref of refs.values()){const bytes=await new StudioAssetService(db).readBytes(u,ref);if(p.cases.some((c:any)=>c.editorialAsset.versionId===ref.versionId))await writeFile(`/tmp/forge-studio-458-prep/creative-followup-2/source-${ref.versionId}.body`,bytes);assets.push({reference:ref,bytes:bytes.length,sha256:createHash('sha256').update(bytes).digest('hex')})}
 await db.$disconnect();await writeFile('/tmp/forge-studio-458-prep/creative-followup-2/asset-preflight.json',JSON.stringify({checks,assets,mutations:0},null,2));console.log('Both exact project/pack snapshots and referenced asset bytes verified; no writes/provider calls')
}
main().catch(e=>{console.error(e.message);process.exitCode=1})
