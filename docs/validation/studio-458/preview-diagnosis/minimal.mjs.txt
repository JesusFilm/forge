import {createServer} from 'node:http';
import {chromium} from '/home/tataihono/.codex/worktrees/8b33/forge/apps/shorts-worker/node_modules/playwright/index.mjs';
const server=createServer((req,res)=>res.end(req.url==='/frame'?'<html><body>Static opaque frame</body></html>':'<html><body><iframe sandbox="allow-scripts" src="/frame"></iframe></body></html>'));
await new Promise(resolve=>server.listen(3590,'127.0.0.1',resolve));
const browser=await chromium.launch({executablePath:'/usr/bin/google-chrome',headless:true});
try {
 const context=await browser.newContext({serviceWorkers:process.argv[2]??'block'}),page=await context.newPage(),errors=[];
 page.on('pageerror',error=>errors.push({message:error.message,stack:error.stack}));
 await page.goto('http://127.0.0.1:3590');await page.waitForTimeout(100);
 console.log(JSON.stringify({mode:process.argv[2]??'block',frameText:await page.frameLocator('iframe').locator('body').innerText(),errors}));
 if(errors.length)process.exitCode=1;
}finally{await browser.close();server.close()}
