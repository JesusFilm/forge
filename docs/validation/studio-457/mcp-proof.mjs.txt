import {readFileSync,writeFileSync} from 'node:fs';
import {createHash,randomUUID} from 'node:crypto';
import assert from 'node:assert/strict';
const tokens=JSON.parse(readFileSync('/tmp/forge-studio-457-probe/oauth-tokens.json','utf8'));
const projectId='762aec64-b7d6-43e1-bb48-7da0e92a82c0';
const evidence=[];
async function rpc(name,args,token=tokens.full){const response=await fetch('http://127.0.0.1:3457/mcp',{method:'POST',headers:{authorization:`Bearer ${token}`,'content-type':'application/json'},body:JSON.stringify({jsonrpc:'2.0',id:randomUUID(),method:'tools/call',params:{name,arguments:args}})});const data=await response.json();return {status:response.status,data,value:data.result?.structuredContent?.result}}
assert.equal((await rpc('studio.read',{projectId},'invalid')).status,401);
const before=await rpc('studio.read',{projectId});assert.equal(before.status,200);
const command={projectId,expectedRevision:before.value.revision,idempotencyKey:randomUUID(),operations:[{kind:'set-metadata',title:'Agent revised'}]};
assert.equal((await rpc('studio.apply',command,tokens.read)).status,403);
assert.equal((await rpc('studio.apply',command)).status,200);
const after=await rpc('studio.read',{projectId});assert.equal(after.value.document.title,'Agent revised');assert.equal(after.value.actor.authority,'delegated');assert.equal(after.value.actor.clientId,'claude-local');
assert.equal((await rpc('studio.apply',{...command,idempotencyKey:randomUUID()})).status,409);
assert.equal((await rpc('studio.activate',{})).data.error.code,-32601);
assert.equal((await rpc('studio.approve',{})).data.error.code,-32601);
const bytes=Buffer.from('export const Component = () => null');
const payload={metadata:{idempotencyKey:randomUUID(),filename:'mcp-component.tsx',mimeType:'text/plain',role:'component',provenance:{status:'recorded',recorded:{}}},byteSize:bytes.length,digest:createHash('sha256').update(bytes).digest('hex')};
assert.equal((await rpc('studio.assetUpload',payload,tokens.read)).status,403);
const grant=await rpc('studio.assetUpload',payload);assert.equal(grant.status,200);
const uploaded=await fetch(grant.value.url,{method:'PUT',body:bytes});assert.equal(uploaded.status,200);const asset=await uploaded.json();
const meta=await rpc('studio.asset',asset.reference);assert.equal(meta.value.actor.authority,'delegated');assert.equal(meta.value.actor.clientId,'claude-local');
const history=await rpc('studio.history',{projectId});
evidence.push({scenario:'same title edit via external scoped OAuth MCP',beforeRevision:before.value.revision,after:after.value,history:history.value,asset:meta.value,negativeChecks:['invalid bearer','read-only edit','read-only upload','stale revision','activation unavailable','approval unavailable'],localDeterministic:true});
writeFileSync('/tmp/forge-studio-457-probe/mcp-evidence.json',JSON.stringify(evidence,null,2));
console.log('External OAuth MCP same-edit, durable component upload and six negative checks passed');
