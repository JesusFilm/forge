import { createServer } from 'node:http';
import { readFileSync,writeFileSync } from 'node:fs';
import { importSPKI,exportJWK,importPKCS8,SignJWT } from 'jose';
const keys=JSON.parse(readFileSync('/tmp/forge-studio-457-probe/local-runtime.json','utf8'));
const jwk=await exportJWK(await importSPKI(keys.publicKey,'EdDSA'));
jwk.kid='studio457';jwk.alg='EdDSA';jwk.use='sig';
const privateKey=await importPKCS8(keys.privateKey,'EdDSA');
const tokens={};
for(const [name,scope] of Object.entries({full:'studio:read studio:edit studio:chat studio:instructions:read',read:'studio:read'})){
 tokens[name]=await new SignJWT({scope,client_id:'claude-local','https://jesusfilm.org/claims/app':'studio-mcp','https://jesusfilm.org/claims/environment':'local'}).setProtectedHeader({alg:'EdDSA',kid:'studio457'}).setSubject('mock-manager-1').setIssuer('http://127.0.0.1:3448').setAudience('http://studio457.localhost:3457/mcp').setIssuedAt().setExpirationTime('2h').sign(privateKey);
}
writeFileSync('/tmp/forge-studio-457-probe/oauth-tokens.json',JSON.stringify(tokens),{mode:0o600});
createServer((req,res)=>{if(req.url!=='/api/auth/jwks'){res.writeHead(404).end();return}res.writeHead(200,{'content-type':'application/json'}).end(JSON.stringify({keys:[jwk]}))}).listen(3448,'127.0.0.1');
