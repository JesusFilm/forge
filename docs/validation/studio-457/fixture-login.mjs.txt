import {readFileSync} from 'node:fs';
const keys=JSON.parse(readFileSync('/tmp/forge-studio-457-probe/local-runtime.json','utf8'));
import http from 'node:http';
import {createHmac} from 'node:crypto';
const encode = value => Buffer.from(JSON.stringify(value)).toString('base64url');
const signingInput = `${encode({alg:'HS256'})}.${encode({id:'mock-manager-1',subject:'mock-manager-1',email:'manager@forge.test',name:'Mock Manager',managerRole:'OPERATOR',scopes:['manager:access'],exp:Math.floor(Date.now()/1000)+86400})}`;
const token = `${signingInput}.${createHmac('sha256',keys.managerSessionSecret).update(signingInput).digest('base64url')}`;
http.createServer((req,res)=>{
 if(req.headers.host !== 'studio457.localhost:3449' || req.url !== '/local-fixture-login'){res.writeHead(404).end();return;}
 res.writeHead(302,{'Set-Cookie':`manager-session=${token}; HttpOnly; SameSite=Lax; Path=/`,Location:'http://studio457.localhost:3457/dashboard/shorts','Cache-Control':'no-store'}).end();
}).listen(3449,'127.0.0.1');
