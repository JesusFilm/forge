import assert from 'node:assert/strict';
import {createRequire} from 'node:module';
import {createHash} from 'node:crypto';
import {writeFileSync} from 'node:fs';
const req=createRequire('/home/tataihono/.codex/worktrees/0941/forge/apps/mastra/package.json');
const {Mastra}=await import(req.resolve('@mastra/core'));
const {Agent}=await import(req.resolve('@mastra/core/agent'));
const {MastraCompositeStore}=await import(req.resolve('@mastra/core/storage'));
const {RequestContext}=await import(req.resolve('@mastra/core/request-context'));
const {MastraEditor}=await import(req.resolve('@mastra/editor'));
const {PostgresStore}=await import(req.resolve('@mastra/pg'));
const {MockLanguageModelV3}=await import(req.resolve('ai/test'));
const captured=[];
const model=new MockLanguageModelV3({doGenerate:async options=>{
 const prompt=options.prompt.filter(m=>m.role==='system').map(m=>m.content).join('\n'); captured.push(prompt);
 return {content:[{type:'text',text:prompt}],finishReason:{unified:'stop',raw:'stop'},usage:{inputTokens:{total:1},outputTokens:{total:1}},warnings:[]};
}});
const pg=new PostgresStore({id:'probe-pg',connectionString:'postgresql://tataihono@127.0.0.1:55457/forge_studio_457_test',schemaName:'mastra'});
const storage=new MastraCompositeStore({id:'probe',default:pg});
const editor=new MastraEditor();
const code=new Agent({id:'studio-probe',name:'Studio probe',instructions:'CODE',model,editor:{instructions:true,tools:false}});
const mastra=new Mastra({agents:{studioProbe:code},editor,storage});
await storage.init();
const blocks=await storage.getStore('promptBlocks'); const agents=await storage.getStore('agents');
await editor.prompt.create({id:'voice',name:'Voice',content:'FIRST {{name}}'});
const b1=await blocks.getLatestVersion('voice');
await editor.prompt.update({id:'voice',activeVersionId:b1.id,status:'published'});
await agents.create({agent:{id:'studio-probe',name:'Studio probe',instructions:[{type:'prompt_block_ref',id:'voice'}],model:{provider:'openai',name:'unused'}}});
const a1=await agents.getLatestVersion('studio-probe');
await editor.agent.update({id:'studio-probe',activeVersionId:a1.id,status:'published'});
const context=new RequestContext([['name','Lyuba']]);
async function run(selection){const selected=await editor.agent.applyStoredOverrides(code,selection,context); return (await selected.generate('echo',{requestContext:context})).text;}
assert.equal(await run({status:'published'}),'FIRST Lyuba');
await editor.prompt.update({id:'voice',content:'SECOND {{name}}'});
const b2=await blocks.getLatestVersion('voice');
assert.equal(await run({status:'published'}),'FIRST Lyuba');
assert.equal(await editor.prompt.preview([{type:'prompt_block_ref',id:'voice'}],{name:'Lyuba'}),'SECOND Lyuba');
const namespaceDraft=await editor.prompt.getById('voice',{status:'draft'});
const directDraft=await blocks.getByIdResolved('voice',{status:'draft'});
assert.equal(namespaceDraft.content,'FIRST {{name}}'); assert.equal(directDraft.content,'SECOND {{name}}');
const pinned=await editor.agent.applyStoredOverrides(code,{versionId:a1.id},context);
await editor.prompt.update({id:'voice',activeVersionId:b2.id,status:'published'});
assert.equal((await pinned.generate('echo',{requestContext:context})).text,'SECOND Lyuba');
await editor.agent.update({id:'studio-probe',instructions:'AGENT DRAFT'});
const a2=await agents.getLatestVersion('studio-probe');
assert.equal(await run(), 'AGENT DRAFT');
assert.equal(await run({status:'published'}),'AGENT DRAFT'); // Native agent.update automatically activates new version!
await editor.agent.update({id:'studio-probe',activeVersionId:a1.id,status:'published'});
assert.equal(await run({status:'published'}),'SECOND Lyuba');
await agents.createVersion({id:crypto.randomUUID(),agentId:'studio-probe',versionNumber:3,name:'Studio probe',instructions:'SAFE DRAFT',model:{provider:'openai',name:'unused'}});
assert.equal(await run({status:'published'}),'SECOND Lyuba');
assert.equal(await run({status:'draft'}),'SAFE DRAFT');
await editor.agent.update({id:'studio-probe',activeVersionId:a2.id,status:'published'});
assert.equal(await run({status:'published'}),'AGENT DRAFT');
await editor.agent.update({id:'studio-probe',activeVersionId:a1.id,status:'published'});
await editor.prompt.update({id:'voice',activeVersionId:b1.id,status:'published'});
assert.equal(await run({status:'published'}),'FIRST Lyuba');
const restored=await blocks.createVersion({id:crypto.randomUUID(),blockId:'voice',versionNumber:3,name:b1.name,content:b1.content,changeMessage:'Restore first into new draft'});
assert.equal((await blocks.getByIdResolved('voice',{status:'draft'})).resolvedVersionId,restored.id);
assert.equal((await blocks.getByIdResolved('voice',{status:'published'})).resolvedVersionId,b1.id);
const results={evidence:'Actual native Editor and Postgres; deterministic local provider double; no paid provider calls',versions:{agent:[a1.id,a2.id],block:[b1.id,b2.id,restored.id]},captured:captured.map(text=>({text,sha256:createHash('sha256').update(text).digest('hex')})),agentUpdateAutoActivates:true,nativeCreateVersionPreservesActivation:true,namespaceDraftIgnored:true,mutableBlockResolvedAfterAgentSelection:true,explicitActivationAndRestore:true};
writeFileSync('/tmp/forge-studio-457-probe/results.json',JSON.stringify(results,null,2)); console.log(JSON.stringify(results,null,2));
await pg.close();
