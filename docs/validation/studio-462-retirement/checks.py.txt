import os,subprocess,json,time
from pathlib import Path
base=Path('/home/tataihono/.local/share/forge/studio-462-retirement')
env={k:os.environ[k] for k in ['PATH','HOME'] if k in os.environ}
env.update(TMPDIR=str(base/'tmp'),CI='1',NEXT_TELEMETRY_DISABLED='1')
checks=[
('worker-full',['pnpm','--filter','@forge/shorts-worker','exec','vitest','run','--maxWorkers=1','--no-file-parallelism']),
('manager-full',['pnpm','--filter','@forge/manager','exec','vitest','run','--maxWorkers=1','--no-file-parallelism']),
('compositions-full',['pnpm','--filter','@forge/shorts-compositions','exec','vitest','run','--maxWorkers=1','--no-file-parallelism']),
('worker-typecheck',['pnpm','--filter','@forge/shorts-worker','typecheck']),
('manager-typecheck',['pnpm','--filter','@forge/manager','typecheck']),
('compositions-typecheck',['pnpm','--filter','@forge/shorts-compositions','typecheck']),
('worker-build',['pnpm','--filter','@forge/shorts-worker','build']),
('devotional-prebundle',['pnpm','--filter','@forge/shorts-worker','prebundle',str(base/'devotional-bundle')]),
]
results=[]
for name,cmd in checks:
 start=time.monotonic()
 with (base/'logs'/f'{name}.log').open('w') as out:
  p=subprocess.run(cmd,env=env,stdout=out,stderr=subprocess.STDOUT)
 results.append(dict(name=name,exit=p.returncode,seconds=round(time.monotonic()-start,3)))
 (base/'evidence/checks.json').write_text(json.dumps(results,indent=2)+'\n')
 if p.returncode: print(name,'FAIL',flush=True)
 else: print(name,'PASS',flush=True)
