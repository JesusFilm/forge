import os,subprocess,time,json,urllib.request,urllib.error
from pathlib import Path
base=Path('/home/tataihono/.local/share/forge/studio-462-retirement')
env={k:os.environ[k] for k in ['PATH','HOME'] if k in os.environ}
env.update(TMPDIR=str(base/'tmp'),NODE_ENV='production',PORT='54662',SHORTS_WORKER_API_KEYS='fixture-worker-key',RAILWAY_S3_ENDPOINT='http://127.0.0.1:1',RAILWAY_S3_REGION='local',RAILWAY_S3_BUCKET='fixture',RAILWAY_S3_ACCESS_KEY_ID='fixture',RAILWAY_S3_SECRET_ACCESS_KEY='fixture',DEVOTIONAL_WORKSPACE_CAPABILITY_ORIGIN='https://workspace.invalid',SHORTS_WORKER_DEVOTIONAL_BUNDLE_DIR=str(base/'devotional-bundle'))
with (base/'logs/worker-http.log').open('w') as out:
 p=subprocess.Popen(['node','dist/server.js'],cwd='apps/shorts-worker',env=env,stdout=out,stderr=subprocess.STDOUT)
 try:
  ready=False
  for _ in range(100):
   if p.poll() is not None: raise RuntimeError('worker exited at startup')
   try:
    with urllib.request.urlopen('http://127.0.0.1:54662/health',timeout=.2) as r: ready=r.status==200
   except OSError: time.sleep(.1)
   if ready: break
  if not ready: raise RuntimeError('worker readiness failed')
  results=[]
  for method,path,body,auth,status in [
   ('GET','/health',None,False,200),
   ('POST','/jobs',{'kind':'prepare'},False,401),
   ('POST','/jobs',{'kind':'prepare','assetId':'fixture','source':{'url':'https://stream.mux.com/fixture.m3u8'},'clip':{'startSec':0,'endSec':10},'transcription':{'language':'en'}},True,400),
   ('POST','/jobs',{'kind':'render','assetId':'fixture'},True,400),
   ('GET','/jobs/unknown',None,False,401),
   ('GET','/jobs/unknown',None,True,404),
   ('DELETE','/jobs/unknown',None,False,401),
   ('DELETE','/jobs/unknown',None,True,404),
  ]:
   headers={'Content-Type':'application/json'}
   if auth: headers['Authorization']='Bearer fixture-worker-key'
   req=urllib.request.Request('http://127.0.0.1:54662'+path,data=json.dumps(body).encode() if body else None,headers=headers,method=method)
   try: response=urllib.request.urlopen(req,timeout=2)
   except urllib.error.HTTPError as error: response=error
   with response:
    got=response.status; data=json.loads(response.read())
   results.append(dict(method=method,path=path,authenticated=auth,expected=status,observed=got,body=data))
   if got!=status: raise RuntimeError('unexpected HTTP outcome')
  (base/'evidence/worker-http.json').write_text(json.dumps(dict(kind='host compiled dist, not OCI image',results=results),indent=2)+'\n')
 finally:
  p.terminate()
  try:p.wait(timeout=5)
  except subprocess.TimeoutExpired:p.kill();p.wait()
